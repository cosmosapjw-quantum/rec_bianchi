281:       fscanf(fA, "%le", &(twog->A2s_tab[b]));
282:       fscanf(fA, "%le", &(twog->A3s3d_tab[b]));
283:       fscanf(fA, "%le", &(twog->A4s4d_tab[b]));

288:    L2s1s_current = 0.;

290:    for (b = 0; b < NSUBLYA; b++) twog->A2s_tab[b] *= L2s1s/L2s1s_current;
291:
292:
293:   /* Switches for the various effects considered in Hirata (2008) and diffusion:
294:       Effect A: correct 2s-->1s rate, with stimulated decays and absorptions of non-thermal photons
295:       Effect B: Sub-Lyman-alpha two-photon decays
296:       Effect C: Super-Lyman-alpha two-photon decays
297:       Effect D: Raman scattering */
